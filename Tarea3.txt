from pyspark.sql import SparkSession
from pyspark.sql import functions as F

# 1. Crear la sesión de Spark
spark = SparkSession.builder \
    .appName("Tarea3_SECOP_Batch") \
    .getOrCreate()

# Configurar nivel de log para evitar mensajes innecesarios
spark.sparkContext.setLogLevel("WARN")

# 2. Ruta del archivo en HDFS
hdfs_path = "hdfs://localhost:9000/Tarea3/rows.csv"

# 3. Cargar el dataset de SECOP desde HDFS
df = spark.read.format("csv") \
    .option("header", "true") \
    .option("inferSchema", "true") \
    .load(hdfs_path)

# 4. Mostrar el esquema y las primeras filas
print("=== ESQUEMA DEL DATASET SECOP ===")
df.printSchema()

print("=== MUESTRA DE DATOS ===")
df.show(5, truncate=False)

print(f"Total registros cargados: {df.count()}")

# Finalizar la sesión de Spark
spark.stop()
